;==============================================================================
;  MODULE FORMAT
;==============================================================================
;
;  Header, 32 bytes (words are big endian):
;    +0   'TFMX'
;    +4   number of frequency sequences - 1
;    +6   number of volume sequences - 1
;    +8   number of patterns - 1
;    +10  number of track steps - 1
;    +12  pattern size in bytes (64 = 32 rows)
;    +14  (not used by this replayer)
;    +16  number of subsongs - 1
;  then, back to back:
;    frequency sequences   64 bytes each
;    volume sequences      64 bytes each (these are the instruments)
;    patterns              <pattern size> bytes each
;    track table           12 bytes per step
;    subsong table         6 bytes per subsong
;    sound effect table    4 bytes per effect (its size is not stored)
;
;  Track step: 12 bytes = 4 bytes for each voice (A at +0, B at +4, C at +8).
;  Each voice moves through the track table on its own: it reads its next
;  step when its own pattern is over.
;    +0  pattern number
;    +1  note transpose (signed, in semitones)
;    +2  instrument transpose (added to the instrument numbers of the notes)
;    +3  effect: $Fx = volume attenuation x for this voice (until the next $Fx)
;                $Ex = song speed: x frames per row
;                other values = no effect
;
;  Pattern row: 2 bytes
;    note  bits 0-6  0 = no note, 1 = end of the pattern, else the note
;                    (index in tfmx_ym_frequencies: 12 = C-2, 24 = C-3...)
;          bit 7     change the pitch only, don't restart the instrument
;    info  bits 0-4  instrument (volume sequence) number
;          bit 5     portamento. The slide speed (signed) is the info byte of
;                    the PREVIOUS row (of the last row for row 0)
;
;  Frequency sequence: one byte per frame
;    $00-$7F   transpose: added to the note and the track transpose
;    $80-$DF   fixed note in bits 0-6, the pattern note is ignored (drums)
;              (in noise only mode the same byte is the noise value, relative
;              to the note or fixed. YM noise period = 31 - (value & 31))
;    $E0,pos   continue at position pos
;    $E1       end: keep the last value
;    $E2       restart the volume sequence, then the next byte
;    $E3,s,d   vibrato speed s and depth d, then a value byte
;    $E4,n     tone + noise, noise value n, then a value byte
;    $E5       noise only, then the next byte
;    $E6       tone only, then a value byte
;    $E7,n     switch to frequency sequence n
;    $E8,n     wait n frames
;    $E9,p     YM hardware envelope: period p, triangle shape. This is the
;              "buzz" sound, heard on channels whose volume has bit 4 set
;
;  Volume sequence (instrument): 64 bytes
;    +0  speed: a new volume byte every 'speed' frames
;    +1  frequency sequence number
;    +2  vibrato speed
;    +3  vibrato depth
;    +4  vibrato delay, in frames
;    +5  volumes 0-15 (bit 4 = YM envelope mode), and the commands:
;        $E0,pos  continue at position pos (counted from +0, header included)
;        $E1      end: keep the last volume
;        $E8,n    wait n frames
;
;  Subsong: 3 words = first track step, last track step, speed
;  Sound effect: 2 words = note (0 = no effect), instrument

;---- header (32 bytes) ----
                                dc.b    'TFMX'                              ; magic
                                dc.w    7-1                                 ; TFMX_HDR_NUM_FSEQ      7 frequency sequences
                                dc.w    8-1                                 ; TFMX_HDR_NUM_VSEQ      8 volume sequences (instruments)
                                dc.w    10-1                                ; TFMX_HDR_NUM_PATTERNS  10 patterns
                                dc.w    56-1                                ; TFMX_HDR_NUM_STEPS     56 track steps
                                dc.w    64                                  ; TFMX_HDR_PATTERN_SIZE  64 bytes = 32 rows
                                dc.w    0                                   ; (not used)
                                dc.w    2-1                                 ; TFMX_HDR_NUM_SUBSONGS  2 subsongs
                                dcb.b   TFMX_HDR_SIZE-18,0                  ; (padding)
;---- frequency sequences: TFMX_SEQ_SIZE (64) bytes each ----
; [nn] = position in the sequence, as used by TFMX_SEQ_LOOP
                                ; octave "blip" every 8 frames (lead)
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_TONE                      ; [01] tone only
                                dc.b    $0C                                 ; [02] transpose +12
                                dc.b    $00,$00,$00,$00,$00,$00,$00         ; [03-09] transpose +0 x 7 frames
                                dc.b    TFMX_SEQ_LOOP,2                     ; [10] loop: continue at [02]
                                dcb.b   TFMX_SEQ_SIZE-12,0                  ; (padding)
                                ; one frame one octave up, then the note (pluck)
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_TONE                      ; [01] tone only
                                dc.b    $0C                                 ; [02] transpose +12
                                dc.b    $00                                 ; [03] transpose +0
                                dc.b    TFMX_SEQ_END                        ; [04] end: keep the last value
                                dcb.b   TFMX_SEQ_SIZE-5,0
                                ; bass drum: noise click, then a fast pitch drop
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_NOISE                     ; [01] noise only
                                dc.b    $93                                 ; [02] fixed noise value $13 (period 12)
                                dc.b    TFMX_FSEQ_TONE                      ; [03] tone only
                                dc.b    $97,$94,$90                         ; [04-06] fixed notes B-2 G#2 E-2
                                dc.b    $8C,$88,$8D,$8C,$8B,$8D,$8E         ; [07-13] fixed notes C-2 G#1 C#2 C-2 B-1 C#2 D-2
                                dc.b    TFMX_SEQ_LOOP,7                     ; [14] loop: continue at [07]
                                dcb.b   TFMX_SEQ_SIZE-16,0
                                ; snare: noise click, then tone+noise falling
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_NOISE                     ; [01] noise only
                                dc.b    $93                                 ; [02] fixed noise value $13 (period 12)
                                dc.b    TFMX_FSEQ_TONE                      ; [03] tone only
                                dc.b    $9E,$9F                             ; [04-05] fixed notes F#3 G-3
                                dc.b    TFMX_FSEQ_TONE_NOISE,$95            ; [06] tone + noise, noise value $95 (period 10)
                                dc.b    $9E                                 ; [08] fixed note F#3
                                dc.b    $9D,$9C,$9B,$9A                     ; [09-12] fixed notes F-3 E-3 D#3 D-3
                                dc.b    TFMX_SEQ_LOOP,9                     ; [13] loop: continue at [09]
                                dcb.b   TFMX_SEQ_SIZE-15,0
                                ; hi-hat: high noise
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_NOISE                     ; [01] noise only
                                dc.b    $9F                                 ; [02] fixed noise value $1F (period 0)
                                dc.b    TFMX_SEQ_END                        ; [03] end: keep the last value
                                dcb.b   TFMX_SEQ_SIZE-4,0
                                ; plain tone
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_TONE                      ; [01] tone only
                                dc.b    $00                                 ; [02] transpose +0
                                dc.b    TFMX_SEQ_END                        ; [03] end: keep the last value
                                dcb.b   TFMX_SEQ_SIZE-4,0                   ; (unused)
                                ; one frame of noise, then tone (bass attack)
                                dc.b    TFMX_FSEQ_RETRIGGER                 ; [00] restart the volume sequence
                                dc.b    TFMX_FSEQ_NOISE                     ; [01] noise only
                                dc.b    $9F                                 ; [02] fixed noise value $1F (period 0)
                                dc.b    TFMX_FSEQ_TONE                      ; [03] tone only
                                dc.b    $00                                 ; [04] transpose +0
                                dc.b    TFMX_SEQ_END                        ; [05] end: keep the last value
                                dcb.b   TFMX_SEQ_SIZE-6,0
;---- volume sequences = instruments: TFMX_SEQ_SIZE (64) bytes each ----
